"""Independent check of the refutation of conjecture L at n = 8."""
from itertools import permutations


def inv(w):
    n = len(w)
    return sum(1 for i in range(n) for j in range(i + 1, n) if w[i] > w[j])


def invdeg(w):
    n = len(w)
    d = [0] * n
    for i in range(n):
        for j in range(i + 1, n):
            if w[i] > w[j]:
                d[i] += 1
                d[j] += 1
    return d


def cut(w, a, b):
    n = len(w)
    return tuple((w[(a + j) % n] - b) % n for j in range(n))


for pi in [(0, 3, 6, 1, 4, 7, 2, 5), (0, 1, 2, 6, 7, 5, 3, 4)]:
    n = len(pi)
    prof = {(a, b): inv(cut(pi, a, b)) for a in range(n) for b in range(n)}
    I = min(prof.values())
    opt = [k for k, v in prof.items() if v == I]
    print("pi =", pi, " I =", I, " bound =", (n - 1) ** 2 // 4,
          " #optimal cuts =", len(opt))
    for k in opt:
        w = cut(pi, *k)
        d = invdeg(w)
        print("   cut", k, "line", w, "deg", d, "max deg", max(d),
              "h", [(n - 1) - 2 * x for x in d])
